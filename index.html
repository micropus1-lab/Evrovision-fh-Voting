window.submitVote = async function() { const message = document.getElementById("voteMessage"); const button = document.getElementById("voteButton"); if (button.disabled) return; button.disabled = true; button.textContent = "ОТПРАВКА..."; try { message.textContent = "Проверяем голосование..."; await getRound(); const voterIdKey = "juniorEurovisionVoterId"; let voterId = localStorage.getItem(voterIdKey); if (!voterId) { voterId = crypto.randomUUID(); localStorage.setItem(voterIdKey, voterId); } const votedRound = localStorage.getItem( "juniorEurovisionVotedRound" ); if (votedRound === String(currentRound)) { message.textContent = "Вы уже проголосовали в этом туре!"; return; } const scores = {}; const selectedCountries = []; for (const point of points) { const select = document.getElementById("score" + point); if (!select) { throw new Error("Не найдено поле для " + point + " баллов"); } const country = select.value; if (!country) { message.textContent = "Распределите все баллы!"; return; } if (selectedCountries.includes(country)) { message.textContent = "Нельзя выбирать одну страну дважды!"; return; } selectedCountries.push(country); scores[String(point)] = country; } message.textContent = "Отправляем ваши голоса..."; await addDoc(collection(db, "votes"), { scores: scores, round: currentRound, voterId: voterId, createdAt: serverTimestamp() }); localStorage.setItem( "juniorEurovisionVotedRound", String(currentRound) ); message.textContent = "Ваши голоса успешно отправлены!"; points.forEach(point => { document.getElementById("score" + point).value = ""; }); } catch (error) { console.error("Ошибка голосования:", error); message.textContent = "Ошибка Firebase: " + (error.code || error.message || "Неизвестная ошибка"); } finally { button.disabled = false; button.textContent = "ОТПРАВИТЬ ГОЛОСА"; } };
